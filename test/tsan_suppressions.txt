# ThreadSanitizer suppression file for Typesense tests.
# These are pre-existing upstream issues not introduced by our fork.
# Syntax: <type>:<function_or_module_pattern>
#
# TSAN with halt_on_error=1 aborts on the first issue found, and the
# specific race it hits is non-deterministic. The suppressions below
# cover all known pre-existing races observed across many CI runs.

# ── sparsepp ──────────────────────────────────────────────────────────
# Data race in sparsepp hash map during concurrent batch_memory_index.
race:spp::sparsegroup*::_sizing

# ── Lock-order-inversions (deadlock suppressions) ────────────────────
# Analytics path: SearchAnalytics::add_internal_event acquires locks in
# reverse order vs other paths.
deadlock:SearchAnalytics::add_internal_event

# CollectionManager::get_collection shared_mutex vs Index lock — search
# paths acquire in inconsistent order.
deadlock:CollectionManager::get_collection

# Collection shared_mutexes during join/reference operations — many
# different methods (get_reference_filter_ids, get_filter_ids,
# reference_populate_sort_mapping, etc.) acquire Collection locks in
# inconsistent order during cross-collection joins.
deadlock:Collection::

# ── Collection/Index data races ──────────────────────────────────────
# Shared vector<vector<art_leaf*>> in Index methods — worker threads
# read while main thread does push_back(). Pre-existing upstream gap.
race:Index::search_wildcard
race:Index::batch_memory_index
race:Index::refresh_schemas

# Concurrent collection loading races on reference_info map and
# AuthManager state during CollectionManager::load.
race:CollectionManager::add_referenced_ins
race:CollectionManager::load_collection
race:CollectionManager::load
race:AuthManager::init

# ── RocksDB internals ────────────────────────────────────────────────
# RocksDB uses lock-free data structures with custom atomics and memory
# barriers that TSAN cannot model. These are well-tested upstream
# components, not bugs in our code. Only RocksDB-internal functions are
# suppressed — our Store/Collection code remains fully instrumented.
#
# MemTable concurrent skiplist (lock-free, custom atomics):
race:rocksdb::MemTable::Add
race:BytewiseComparatorImpl::Compare
race:rocksdb::Arena::AllocateNewBlock
race:rocksdb::Arena::AllocateAligned
# BlockBuilder vs ZSTD compression during flush:
race:rocksdb::BlockBuilder::Add
race:ZSTD_compressBlock
# Internal compaction/flush thread pool:
race:rocksdb::CompactionJob
race:rocksdb::FlushJob
race:rocksdb::DBImpl::BackgroundFlush
race:rocksdb::DBImpl::BackgroundCompaction
# Posix file I/O during concurrent open/close:
race:rocksdb::PosixRandomAccessFile
race:rocksdb::PosixSequentialFile
race:rocksdb::PosixWritableFile
